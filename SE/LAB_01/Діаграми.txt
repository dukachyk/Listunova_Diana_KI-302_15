# Система бронювання та купівлі авіаквитків

UML діаграма сценарію роботи клієнтської частини


@startuml

left to right direction
actor "Клієнт"
actor "Сервер"

Клієнт --> (Додавання авіарейсу)
Клієнт --> (Додавання квитків до авіарейсу)
Клієнт --> (Пошук авіарейсу)
Клієнт --> (Перегляд інформації про авіарейс)
Клієнт --> (Оновлення інформації про авіарейс)
Клієнт --> (Купівля авіаквитків)
Клієнт --> (Бронювання авіаквитків)
Клієнт --> (Пошук найефективнішого маршруту)

(Додавання авіарейсу) --> Сервер
(Додавання квитків до авіарейсу) --> Сервер
(Пошук авіарейсу) --> Сервер
(Перегляд інформації про авіарейс) --> Сервер
(Оновлення інформації про авіарейс) --> Сервер
(Купівля авіаквитків) --> Сервер
(Бронювання авіаквитків) --> Сервер
(Пошук найефективнішого маршруту) --> Сервер

@enduml


UML діаграма сценарію роботи серверної частини

@startuml
left to right direction
actor "Сервер" as Server

package "Передача даних до клієнта" {
    (Інформація про авірейс)
    (Інформація про квиток)
    (Інформація про купівлю)
    (Інформація про бронювання)
}

package "Сервер" {
    (Обробка запиту)
    (Перевірка правильного\nприйнятого пакету)
    (Декодування команди \nприйнятої від клієнта)
}

package "База даних" {
	(Додавання авіарейсу)
	(Додавання квитків до авіарейсу)
	(Пошук авіарейсу)
	(Перегляд інформації про авіарейс)
	(Оновлення інформації про авіарейс)
	(Купівля авіаквитків)
	(Бронювання авіаквитків)
}

Server --> (Інформація про авірейс)
Server --> (Інформація про квиток)
Server --> (Інформація про купівлю)
Server --> (Інформація про бронювання)
Server --> (Обробка запиту)
(Обробка запиту) --> (Декодування команди \nприйнятої від клієнта)
(Обробка запиту) --> (Перевірка правильного\nприйнятого пакету)
(Перевірка правильного\nприйнятого пакету) --> (Обробка запиту)

(Декодування команди \nприйнятої від клієнта) --> (Додавання авіарейсу)
(Декодування команди \nприйнятої від клієнта) --> (Додавання квитків до авіарейсу)
(Декодування команди \nприйнятої від клієнта) --> (Пошук авіарейсу)
(Декодування команди \nприйнятої від клієнта) --> (Перегляд інформації про авіарейс)
(Декодування команди \nприйнятої від клієнта) --> (Оновлення інформації про авіарейс)
(Декодування команди \nприйнятої від клієнта) --> (Купівля авіаквитків)
(Декодування команди \nприйнятої від клієнта) --> (Бронювання авіаквитків)
@enduml



Клієнтська частина процес додавання авіарейсу

@startuml
start
repeat :Введення дати вильоту;
:Введення точки вильоту;
:Введення точки прильоту;
:Введення часу польоту;
:Запит на сервер;
backward:Повідомлення про\nпомилку додавання;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Додавання успішне;
stop
@enduml


Серверна  частина процес додавання квитків до авіарейсу

@startuml
start

:Отримання даних<

if (Дані верифіковані) then (Так)
    :Запис кількості, ціни та класу квитків в БД;
    :Успішне додавання;
else (Ні)
	:Помилка верифікації даних;
endif
stop
@enduml


Клієнтська частина процес пошуку авіарейсу

@startuml
start

repeat :Введення точки вильоту;
:Введення точки прильоту;
fork
	:Введення часу польоту;
fork again
	:Введення дати вильоту;
end fork

:Запит на сервер;
backward:Повідомлення про\nпомилку пошуку авіарейсу;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Вивід авіарейсів;
stop
@enduml


Клієнтська частина процес перегляду розширеної інформації про авіарейс

@startuml
start

repeat:Вибір авіарейсу;

:Запит на сервер для отримання базової інформації про авіарейс;
backward:Повідомлення про\nпомилку пошуку авіарейсу;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Вивід базової інформації про авіарейс;
repeat :Запит на сервер для отримання інформації про квитки для авіарейсу;
backward:Повідомлення про\nпомилку отримання інформації про квитки для авіарейсу;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Вивід інформації про квитки для авіарейсу;
stop
@enduml


Клієнтська частина процес оновлення авіарейсу

@startuml
start
repeat :Вибір авіарейсу для редагуванння;
if (Авіарейс існує?) then (Так)
    fork
		:Введення нової\nдати вильоту;
	fork again
		:Введення нової\nточки вильоту;
	fork again
		:Введення нової\nточки прильоту;
	fork again
		:Введення нового\nчасу польоту;
	end fork
	:Запит на сервер для оновлення даних авіарейсу;
else (Ні)
	:Помилка -- авіарейс не існує;
endif


backward:Повідомлення про\nпомилку додавання;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)
:Додавання успішне;
stop
@enduml


Серверна  частина процес оновлення квитків для авіарейсі

@startuml
start

:Отримання даних<

if (Дані верифіковані) then (Так)
    fork
		:Оновлення кількості\nквитків в БД;
	fork again
		:Оновлення ціни \nквитків в БД;
	fork again
		:Оновлення класу\nквитків в БД;
	fork again
		:Оновлення кількості\nквитків в БД;
	end fork
else (Ні)
	:Помилка верифікації даних;
endif
stop
@enduml


Серверна  частина процес купівлі авіаквитків

@startuml
start

:Отримання даних<

if (Дані верифіковані) then (Так)
    :Додавання купленого квитка в БД;
    :Оновлення кількості\n вільних квитків в БД;
    
else (Ні)
	:Помилка верифікації даних;
endif
stop
@enduml



Серверна  частина процес бронювання авіаквитків

@startuml
start

:Отримання даних<

if (Дані верифіковані) then (Так)
    :Додавання заброньованого квитка в БД;
    :Оновлення кількості\n вільних квитків в БД;
    
    while (Час бронювання не вичерпано?)
        :Очікування;
    endwhile

    :Зняття бронювання;
    :Оновлення кількості\n вільних квитків в БД;
    
else (Ні)
	:Помилка верифікації даних;
endif

stop
@enduml



Клієнтська частина процес пошуку найефективнішого маршруту

@startuml
start

repeat :Введення точки вильоту;
:Введення точки прильоту;
:Введення дати вильоту;

:Запит на сервер для пошуку прямих рейсів;
backward:Повідомлення про\nпомилку пошуку;
repeat while (Чи надійшла відповідь?) is (Ні) not (Так)

if (Прямий рейс знайдено?) then (Так)
    :Вивід прямого рейсу;
else (Ні)
    :Запит на сервер для пошуку можливих рейсів з пересадкою;
	if (Чи надійшла відповідь?) then (Так)
		:Вивід маршрутів з пересадками в\nпорядку зростання часу польоту;
		if (Рейси з пересадками знайдено?) then (Так)
			:Вивід маршрутів з пересадками в\nпорядку зростання часу польоту;
		else (Ні)
			:Повідомлення про відсутність маршрутів;
		endif
	else (Ні)
		:Повідомлення про відсутність маршрутів;
	endif
endif

stop
@enduml
